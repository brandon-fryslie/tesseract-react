```typescript
import { observable, makeObservable } from 'mobx';

export default class ClipModel extends BaseModel {

    makeObservable(this);
    this.displayName = displayName;
    this.clipId = clipId;
    this.controls = controls;
  }
```

```typescript
import { observable, makeObservable } from 'mobx';
import BaseModel from './BaseModel';

type ControlType = 'knob' | 'slider' | '2-axis-slider' | 'text-input' | 'file-picker';

export default class ControlModel extends BaseModel {
  @observable displayName!: string;
  @observable type!: ControlType;
  @observable defaultValue!: number | string;
  @observable currentValue!: number | string;
  @observable fieldName!: string;
  @observable minValue!: number;
  @observable maxValue!: number;

  constructor(
    displayName: string,
    type: ControlType,
    defaultValue: number | string,
    currentValue: number | string | null,
    fieldName: string,
    maxValue: number = 1,
    minValue: number = 0
  ) {
    super();
    makeObservable(this);

    this.displayName = displayName;
    this.type = type;
    this.defaultValue = defaultValue;
    this.currentValue = currentValue ?? defaultValue;
    this.fieldName = fieldName;
    this.minValue = minValue;
    this.maxValue = maxValue;
  }

  toJS(): Record<string, any> {
    return {
      displayName: this.displayName,
      type: this.type,
      defaultValue: this.defaultValue,
      currentValue: this.currentValue,
      fieldName: this.fieldName,
      minValue: this.minValue,
      maxValue: this.maxValue,
    };
  }

  static fromJS(obj: Record<string, any>): ControlModel {
    return new ControlModel(
      obj.displayName,
      obj.type,
      obj.defaultValue,
      obj.currentValue,
      obj.fieldName,
      obj.maxValue,
      obj.minValue
    );
  }
}
```

```typescript
import { observable, makeObservable, IObservableArray } from 'mobx';

export default class BaseStore<T extends { id: string | number }> {
  @observable items: IObservableArray<T> = observable.array([]);

  static instance: any; // Will be overridden in subclasses

  constructor() {
    makeObservable(this);
  }

  static get<T extends BaseStore<any>>(): T {
    if (this.instance == null) {
      this.instance = new this();
    }
    return this.instance as T;
  }

  abstract getModelType(): { fromJS(obj: any): T };

  getItems(): T[] {
    return this.items.slice().sort((p1, p2) => {
      return p1.id < p2.id ? -1 : p1.id > p2.id ? 1 : 0;
    });
  }

  addItem(item: T): void {
    this.items.push(item);
  }

  removeItem(item: T): void {
    const itemToRemove = this.items.find(i => i.id === item.id);
    if (itemToRemove == null) {
      throw new Error(`[BaseStore] Could not find item to remove: ${item.id}`);
    }
    this.items.remove(itemToRemove);
  }

  find<K extends keyof T>(key: K, value: T[K]): T {
    const item = this.items.find(i => i[key] === value);
    if (item == null) {
      throw new Error(`Could not find item with ${String(key)}: ${value}`);
    }
    return item;
  }

  refreshFromJS(arr: any[]): void {
    this.items.replace(arr.map(item => this.getModelType().fromJS(item)));
  }
}
```

```typescript
import React from 'react';
import ListGroup from 'react-bootstrap/ListGroup';
import Card from 'react-bootstrap/Card';
import { observer } from 'mobx-react';
import PlaylistModel from '../models/PlaylistModel';

interface PlaylistsListProps {
  onItemClick: (event: React.MouseEvent, playlist: PlaylistModel) => void;
  activePlaylist?: PlaylistModel | null;
  items: PlaylistModel[];
}

@observer
class PlaylistsList extends React.Component<PlaylistsListProps> {
  render() {
    const { activePlaylist, items, onItemClick } = this.props;

    const sortedItems = items.slice().sort((p1, p2) => {
      return p1.id < p2.id ? -1 : p1.id > p2.id ? 1 : 0;
    });

    return (
      <Card className="mt-3 mb-3">
        <Card.Header>Playlists</Card.Header>
        <ListGroup as="ul">
          {sortedItems.map((playlist, idx) => (
            <ListGroup.Item
              action
              as="li"
              key={idx}
              onClick={(e) => onItemClick(e, playlist)}
              active={activePlaylist?.id === playlist.id}
            >
              {playlist.displayName}
            </ListGroup.Item>
          ))}
        </ListGroup>
      </Card>
    );
  }
}

export default PlaylistsList;
```
