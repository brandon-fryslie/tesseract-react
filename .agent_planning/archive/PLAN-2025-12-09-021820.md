```typescript
import { observable, action, makeObservable, IObservableArray } from 'mobx';
import BaseModel from '../models/BaseModel';

export default abstract class BaseStore<T extends { id: string | number }> {
  @observable items: IObservableArray<T> = observable.array([]);

  static instance: any; // Will be overridden in subclasses

  constructor() {
    makeObservable(this);
  }

  static get<S extends BaseStore<any>>(): S {
    if (this.instance == null) {
      this.instance = new this();
    }
    return this.instance as S;
  }

  abstract getModelType(): { fromJS(obj: any): T };

  getItems(): T[] {
    return this.items.slice().sort((p1, p2) => {
      return p1.id < p2.id ? -1 : p1.id > p2.id ? 1 : 0;
    });
  }

  @action
  addItem(item: T): void {
    this.items.push(item);
  }

  @action
  removeItem(item: T): void {
    const itemToRemove = this.items.find(i => i.id === item.id);
    if (itemToRemove == null) {
      throw new Error(`[BaseStore] Could not find item to remove: ${item.id}`);
    }
    this.items.remove(itemToRemove);
  }

  find<K extends keyof T>(key: K, value: T[K]): T {
    const item = this.items.find(i => i[key] === value);
    if (item == null) {
      throw new Error(`Could not find item with ${String(key)}: ${value}`);
    }
    return item;
  }

  @action
  refreshFromJS(arr: any[]): void {
    this.items.replace(arr.map(item => this.getModelType().fromJS(item)));
  }
}
```

Critical points:
- Generic constraint requires `id` property
- Use `IObservableArray<T>` for proper MobX array typing
- Singleton pattern needs careful typing (instance is any)
- Abstract method enforces subclass implementation
- MobX 6 array methods (`.remove()`, `.replace()`) need verification

## Rollback Strategy

### Phase-Level Rollback

Each phase can be rolled back independently:

1. **Revert files** to .js/.jsx extensions
2. **Remove type annotations** (git diff shows changes)
3. **Restore original imports**
4. **Verify functionality** with tests

### Full Migration Rollback

If entire migration needs rollback:

1. **Git revert** to commit before Phase 0
2. **Remove TypeScript packages**:
   ```bash
   pnpm remove typescript @types/react @types/react-dom @types/node
   pnpm remove @typescript-eslint/parser @typescript-eslint/eslint-plugin
   ```
3. **Delete tsconfig.json**
4. **Revert babel.config.js** (remove TypeScript preset)
5. **Revert webpack config** (remove .ts/.tsx extensions)
6. **Revert .eslintrc** (remove TypeScript overrides)
7. **Run tests** to verify rollback successful

### Partial Migration State

The migration is designed to support partial completion:
- `allowJs: true` allows .js and .ts files to coexist
- Can pause after any phase with working application
- Can deploy partially migrated code if needed

## Appendix: Key Code Examples

### MobX Model Pattern
```typescript
import { observable, makeObservable } from 'mobx';
import BaseModel from './BaseModel';

export default class ExampleModel extends BaseModel {
  @observable name!: string;
  @observable count!: number;

  constructor(name: string, count: number) {
    super();
    makeObservable(this);
    this.name = name;
    this.count = count;
  }

  toJS(): IExampleData {
    return { name: this.name, count: this.count };
  }

  static fromJS(data: IExampleData): ExampleModel {
    return new ExampleModel(data.name, data.count);
  }
}
```

### MobX Store Pattern
```typescript
import { observable, action, makeObservable } from 'mobx';
import BaseStore from './BaseStore';
import ExampleModel from '../models/ExampleModel';

export default class ExampleStore extends BaseStore<ExampleModel> {
  constructor() {

    makeObservable(this);
```

### React Component Pattern (Class with Observer)
```typescript
import React from 'react';
import { observer } from 'mobx-react';
import ExampleStore from '../stores/ExampleStore';

interface ExampleComponentProps {
  title: string;
}

interface ExampleComponentState {
  filter: string;
}

@observer
class ExampleComponent extends React.Component<ExampleComponentProps, ExampleComponentState> {
  private store: ExampleStore;

  constructor(props: ExampleComponentProps) {
    super(props);
    this.store = ExampleStore.get();
    this.state = { filter: '' };
  }

  handleFilterChange = (event: React.ChangeEvent<HTMLInputElement>): void => {
    this.setState({ filter: event.target.value });
  };

  render(): React.ReactNode {
    const { title } = this.props;
    const { filter } = this.state;
    const items = this.store.getItems().filter(item =>
      item.name.includes(filter)
    );

    return (
      <div>
        <h1>{title}</h1>
        <input value={filter} onChange={this.handleFilterChange} />
        {items.map(item => <div key={item.id}>{item.name}</div>)}
      </div>
    );
  }
}

export default ExampleComponent;
```
